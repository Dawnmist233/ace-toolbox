# ---------- 安装 ACE（含启用预启动模式）----------
install|findsetup|ACE-Setup64.exe|Uninstaller.exe
install|setup|install,-install,/install
install|extract|C:\Program Files\AntiCheatExpert
install|run|C:\Program Files\AntiCheatExpert\ACE-Tray.exe|start

# ---------- 启动 / 激活（免重启）----------
start|scfilter|AntiCheatExpert,ACE-|ACE_|ace-game|SGuard|TenioDL
start|scstart
start|extract|C:\Program Files\AntiCheatExpert
start|run|C:\Program Files\AntiCheatExpert\ACE-Tray.exe|start

# ---------- 静默关闭 ----------
close|kill|ACE-Tray.exe,ACE-TrayV.exe,ACE-Service64.exe,SGuard64.exe,SGuardSvc64.exe,SGuardSvc32.exe,ACE-BOOT.exe,ACE-Guard Service,ACE-Guard Client,^ACE-,^ACE_ ,^SGuard,^AntiCheat
close|scfilter|AntiCheatExpert,ACE-|ACE_|ace-game|SGuard|TenioDL
close|scstop

# ---------- 彻底卸载 ----------
uninstall|kill|ACE-Tray.exe,ACE-TrayV.exe,ACE-Service64.exe,SGuard64.exe,SGuardSvc64.exe,SGuardSvc32.exe,ACE-BOOT.exe,ACE-Guard Service,ACE-Guard Client,^ACE-,^ACE_ ,^SGuard,^AntiCheat
uninstall|scfilter|AntiCheatExpert,ACE-|ACE_|ace-game|SGuard|TenioDL
uninstall|scdel
uninstall|deldir|C:\Program Files\AntiCheatExpert
uninstall|deldir|C:\Program Files (x86)\AntiCheatExpert
uninstall|deldir|C:\ProgramData\AntiCheatExpert
uninstall|regdel|HKLM|SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall\AntiCheatExpert
uninstall|regdel|HKLM|SOFTWARE\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall\AntiCheatExpert
uninstall|regdel|HKCU|SOFTWARE\AppDataLow\AntiCheatExpert

# ============================================================
#  v2.0.6 新增：关闭 ACE（云下发）

# ============================================================
disable|kill|ACE-Tray.exe,ACE-TrayV.exe,ACE-Service64.exe,SGuard64.exe,SGuardSvc64.exe,SGuardSvc32.exe,ACE-BOOT.exe,ACE-Guard Service,ACE-Guard Client,^ACE-,^ACE_ ,^SGuard,^AntiCheat

disable|scdisable
